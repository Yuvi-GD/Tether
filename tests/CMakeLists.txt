if(TETHER_BUILD_TESTS)
    add_executable(tether_behavior_test unit/behavior_test.c)
    target_link_libraries(tether_behavior_test PRIVATE tether_runtime)
    foreach(scenario IN ITEMS layout visibility input lifetime text widgets menu_interaction event_contract disabled)
        add_test(NAME behavior_${scenario} COMMAND tether_behavior_test ${scenario})
        set_tests_properties(behavior_${scenario} PROPERTIES LABELS "correctness;unit" TIMEOUT 15)
    endforeach()
    add_executable(tether_ecs_test unit/ecs_test.c)
    target_link_libraries(tether_ecs_test PRIVATE tether_runtime)
    add_test(NAME ecs_test COMMAND tether_ecs_test)
    set_tests_properties(ecs_test PROPERTIES LABELS "correctness;unit")
    add_subdirectory(integration)
endif()
